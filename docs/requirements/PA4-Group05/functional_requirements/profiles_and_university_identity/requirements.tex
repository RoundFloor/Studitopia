\hypertarget{sec:us-08-manage-profile}{}
\addcontentsline{toc}{subsubsection}{US-08: Manage Profile}
\subsubsection*{US-08: Manage Profile}
\par\medskip

\textbf{As a} university student, \textbf{I want to} create and update my profile, \textbf{so that} other users can understand my academic background and interests.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.17\textwidth}|>{\raggedright\arraybackslash}p{0.53\textwidth}|>{\raggedright\arraybackslash}p{0.11\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-08.1 (Profile Update)} & The system shall allow a user to create and update their profile. & UI, Frontend, Backend & Partial \\ \hline
\textbf{FR-08.2 (Profile Fields)} & The profile shall include display name, username, avatar, short bio, university, major, and academic year. & UI, Frontend & Complete \\ \hline
\textbf{FR-08.3 (Academic Selection)} & The system shall allow users to select their university and major from the supported catalog. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-08.4 (Avatar Validation)} & The system shall validate avatar uploads for file type and size. & Frontend, Backend & Complete \\ \hline
\textbf{FR-08.5 (Profile Persistence)} & The system shall persist profile changes and reflect them across the user's active sessions. & Backend, Database & Complete \\ \hline
\textbf{FR-08.6 (Public Academic Info)} & The system shall display the user's university, major, and academic year on their public forum profile. & UI, Frontend & Not implemented \\ \hline
\end{longtable}
\endgroup

\subsubsection*{US-09: Set Learning Preferences}
\par\medskip

\textbf{As a} university student, \textbf{I want to} specify my courses, subjects, interests, and learning goals, \textbf{so that} the platform can recommend relevant content.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.16\textwidth}|>{\raggedright\arraybackslash}p{0.54\textwidth}|>{\raggedright\arraybackslash}p{0.11\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-09.1 (Course Preferences)} & The system shall allow a university student to specify their current courses. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-09.2 (Subject Preferences)} & The system shall allow a university student to specify subjects and topics of interest. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-09.3 (Goal Preferences)} & The system shall allow a university student to specify learning goals and preferred study times. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-09.4 (Recommendation Use)} & The system shall use these preferences to recommend forum posts, materials, study partners, and optional rooms. & Backend & Not implemented \\ \hline
\textbf{FR-09.5 (Preference Updates)} & The system shall allow users to update their preferences at any time. & UI, Frontend, Backend & Not implemented \\ \hline
\end{longtable}

\hypertarget{sec:us-10-manage-privacy}{}
\addcontentsline{toc}{subsubsection}{US-10: Manage Privacy}
\subsubsection*{US-10: Manage Privacy}
\par\medskip

\textbf{As a} user, \textbf{I want to} control the visibility of my personal information, \textbf{so that} I can protect my privacy while using the platform.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.13\textwidth}|>{\raggedright\arraybackslash}p{0.59\textwidth}|>{\raggedright\arraybackslash}p{0.09\textwidth}|>{\raggedright\arraybackslash}p{0.09\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-10.1 (Visibility Controls)} & The system shall allow a user to control the visibility of personal information such as profile details, study history, and contact information. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-10.2 (Search Opt-Out)} & The system shall allow a user to opt out of appearing in public discovery and study-partner search results. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-10.3 (Online Status)} & The system shall allow a user to hide their online status and last-active time. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-10.4 (Privacy Enforcement)} & The system shall respect privacy settings when displaying user information in forum posts, comments, and member lists. & Backend, Database & Not implemented \\ \hline
\end{longtable}
\endgroup
