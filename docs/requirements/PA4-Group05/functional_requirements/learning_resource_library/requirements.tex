\subsubsection*{US-25: Find Learning Resources}
\par\medskip

\textbf{As a} university student, \textbf{I want to} search for books and learning resources, \textbf{so that} I can find materials related to my courses.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.20\textwidth}|>{\raggedright\arraybackslash}p{0.49\textwidth}|>{\raggedright\arraybackslash}p{0.12\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-25.1 (Resource Search)} & The system shall allow a user to search for books and learning resources. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-25.2 (Resource Search Fields)} & The system shall support search by title, author, subject tags, and description. & Frontend, Backend & Not implemented \\ \hline
\textbf{FR-25.3 (Resource Filtering)} & The system shall support filtering by resource format and subject field. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-25.4 (Resource Ratings)} & The system shall display resource ratings and review counts when available. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-25.5 (Related Resources)} & The system shall display resources related to the user's university, major, and courses. & Backend & Not implemented \\ \hline
\end{longtable}

\subsubsection*{US-26: Receive Resource Recommendations}
\par\medskip

\textbf{As a} university student, \textbf{I want to} receive recommendations for books and learning resources, \textbf{so that} I can discover useful materials.
\par\medskip

\begingroup
\small

\renewcommand{\arraystretch}{1.15}

\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-26.1 (Resource Suggestions)} & The system shall recommend books or learning resources related to the user's subjects, courses, and interests. & Backend & Not implemented \\ \hline
\textbf{FR-26.2 (Recommendation Display)} & The system shall display recommended resources on the user's home or library page. & UI, Frontend & Not implemented \\ \hline
\textbf{FR-26.3 (Save Recommendation)} & The system shall allow users to save recommended resources to a personal list. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-26.4 (Cold Start)} & The system shall provide curated foundational resources for new users without historical activity. & Backend & Not implemented \\ \hline

\addcontentsline{toc}{subsubsection}{US-27: Share Learning Resources}
\subsubsection*{US-27: Share Learning Resources}
\par\medskip

\textbf{As a} university student, \textbf{I want to} share learning resources, \textbf{so that} I can help others access useful materials.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}

\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-27.1 (Publish Resource)} & The system shall allow a user to publish a learning resource to the library. & UI, Frontend, Backend & Partial \\ \hline
\textbf{FR-27.2 (Resource Details)} & The system shall allow the user to provide a title, description, subject tags, and course tags. & UI, Frontend & Not implemented \\ \hline
\textbf{FR-27.3 (Resource File Validation)} & The system shall validate uploaded files for type and size before publishing. & Frontend, Backend & Partial \\ \hline
\textbf{FR-27.4 (Resource Searchability)} & The system shall make published resources searchable after publication. & Backend, Database & Not implemented \\ \hline
\textbf{FR-27.5 (Resource Access)} & The system shall allow other users to view and access published resources according to privacy settings. & Backend & Not implemented \\ \hline
\end{longtable}
