\subsubsection*{US-30: Find a Study Partner}
\par\medskip

\textbf{As a} university student, \textbf{I want to} find users with similar courses and learning goals, \textbf{so that} I can establish useful academic connections.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.18\textwidth}|>{\raggedright\arraybackslash}p{0.53\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-30.1 (Study Partner Discovery)} & The system shall allow a user to find other university students with similar courses, subjects, or learning goals. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-30.2 (Partner Search and Filter)} & The system shall support search and filtering by subject, goal, learning pace, and language. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-30.3 (Profile Cards)} & The system shall display concise profile cards with mutual academic interests. & UI, Frontend & Not implemented \\ \hline
\textbf{FR-30.4 (Mutual Connections)} & The system shall display mutual friends or shared forum contexts when available. & UI, Frontend, Backend & Not implemented \\ \hline
\end{longtable}
\endgroup

\hypertarget{sec:us-31-send-friend-request}{}
\addcontentsline{toc}{subsubsection}{US-31: Send Friend Request}
\subsubsection*{US-31: Send Friend Request}
\par\medskip

\textbf{As a} university student, \textbf{I want to} send a friend request, \textbf{so that} we can maintain a connection for future learning activities.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.23\textwidth}|>{\raggedright\arraybackslash}p{0.42\textwidth}|>{\raggedright\arraybackslash}p{0.16\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-31.1 (Send Request)} & The system shall allow a user to send a friend request to another user. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-31.2 (Duplicate Request Prevention)} & The system shall prevent duplicate friend requests between the same users. & Frontend, Backend, Database & Not implemented \\ \hline
\textbf{FR-31.3 (Request Note)} & The system shall allow the sender to attach a short introductory note. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-31.4 (Request Notification)} & The system shall notify the recipient of the friend request. & Backend & Not implemented \\ \hline
\end{longtable}
\endgroup
\par\medskip

\hypertarget{sec:us-32-manage-friend-requests}{}
\addcontentsline{toc}{subsubsection}{US-32: Manage Friend Requests}
\subsubsection*{US-32: Manage Friend Requests}
\par\medskip

\textbf{As a} user, \textbf{I want to} accept or reject incoming friend requests, \textbf{so that} I can control my social connections.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.19\textwidth}|>{\raggedright\arraybackslash}p{0.52\textwidth}|>{\raggedright\arraybackslash}p{0.11\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-32.1 (Request Center)} & The system shall provide a centralized view of incoming friend requests. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-32.2 (Accept Request)} & The system shall allow a user to accept a friend request and add the sender to their friend list. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-32.3 (Reject Request)} & The system shall allow a user to reject a friend request. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-32.4 (Block Future Requests)} & The system shall allow a user to block future requests from a rejected sender. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-32.5 (Sender Notification)} & The system shall notify the sender when a request is accepted or rejected. & Backend & Not implemented \\ \hline
\end{longtable}
\endgroup

\hypertarget{sec:us-33-create-and-manage-friend-groups}{}
\addcontentsline{toc}{subsubsection}{US-33: Create and Manage Friend Groups}
\subsubsection*{US-33: Create and Manage Friend Groups}
\par\medskip

\textbf{As a} university student, \textbf{I want to} create and manage friend groups, \textbf{so that} I can collaborate privately with classmates and project teams.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.21\textwidth}|>{\raggedright\arraybackslash}p{0.50\textwidth}|>{\raggedright\arraybackslash}p{0.11\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-33.1 (Group Creation)} & The system shall allow a user to create a private or public friend group. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-33.2 (Group Setup)} & The system shall allow the group owner to name the group, set privacy, and invite initial members. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-33.3 (Group Sharing)} & The system shall allow group members to send messages and share documents within the group. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-33.4 (Member List and Invitation)} & The system shall allow members to view the group member list and invite new members. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-33.5 (Owner Member Removal)} & The system shall allow the group owner to remove members when necessary. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-33.6 (Leave Group)} & The system shall allow members to leave a group. & UI, Frontend, Backend & Not implemented \\ \hline
\end{longtable}
\endgroup
