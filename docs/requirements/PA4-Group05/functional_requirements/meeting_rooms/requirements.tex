\subsubsection*{US-34: Receive Room Recommendations}
\par\medskip

\textbf{As a} university student, \textbf{I want to} receive optional study-room recommendations, \textbf{so that} I can find a room when I need live collaboration.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.23\textwidth}|>{\raggedright\arraybackslash}p{0.48\textwidth}|>{\raggedright\arraybackslash}p{0.11\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-34.1 (Optional Room Recommendations)} & The system shall provide optional recommendations for study rooms. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-34.2 (Recommendation Factors)} & Room recommendations shall consider the user's courses, interests, and preferred study times. & Backend & Not implemented \\ \hline
\textbf{FR-34.3 (Room Availability)} & The system shall display room availability and capacity. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-34.4 (Dismiss Room Recommendation)} & The system shall allow users to dismiss irrelevant room suggestions. & UI, Frontend, Backend & Not implemented \\ \hline
\end{longtable}
\endgroup

\hypertarget{sec:us-35-join-or-create-a-study-room}{}
\addcontentsline{toc}{subsubsection}{US-35: Join or Create a Study Room}
\subsubsection*{US-35: Join or Create a Study Room}
\par\medskip

\textbf{As a} university student, \textbf{I want to} join or create a study room, \textbf{so that} I can collaborate with classmates in real time when needed.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.17\textwidth}|>{\raggedright\arraybackslash}p{0.55\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-35.1 (Browse Public Rooms)} & The system shall allow a user to browse available public study rooms. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-35.2 (Join Public Room)} & The system shall allow a user to join a public room with available capacity. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-35.3 (Request Private Room)} & The system shall allow a user to request access to a private or custom room. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-35.4 (Manage Join Requests)} & The system shall allow a room owner or manager to accept or reject join requests. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-35.5 (Create Room)} & The system shall allow a user to create a study room with a name, description, privacy level, and supported tools. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-35.6 (Configure Room)} & The system shall allow room managers to configure capacity, duration, and communication rules. & UI, Frontend, Backend & Not implemented \\ \hline
\end{longtable}
\endgroup

\subsubsection*{US-36: Use Shared Study Tools}
\par\medskip

\textbf{As a} university student, \textbf{I want to} use shared study tools in a room, \textbf{so that} I can collaborate on academic problems with other participants.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.18\textwidth}|>{\raggedright\arraybackslash}p{0.53\textwidth}|>{\raggedright\arraybackslash}p{0.11\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-36.1 (Shared Whiteboard)} & The system shall provide a shared whiteboard for room participants. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-36.2 (Shared Notes)} & The system shall provide shared notes for room participants. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-36.3 (Shared Timer)} & The system shall provide a shared timer for room participants. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-36.4 (Tool Synchronization)} & The system shall synchronize shared tool updates for all room participants. & Frontend, Backend & Not implemented \\ \hline
\textbf{FR-36.5 (Whiteboard Export)} & The system shall allow users to export whiteboard content when supported. & Frontend & Not implemented \\ \hline
\textbf{FR-36.6 (Reconnection Recovery)} & The system shall persist shared notes during the room session and handle reconnection without data loss. & Frontend, Backend & Not implemented \\ \hline
\end{longtable}

\hypertarget{sec:us-37-request-learning-help}{}
\addcontentsline{toc}{subsubsection}{US-37: Request Learning Help}
\subsubsection*{US-37: Request Learning Help}
\par\medskip

\textbf{As a} university student, \textbf{I want to} request learning help, \textbf{so that} I can receive assistance from friends, room participants, or concept learners.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}
\begin{longtable}{|>{\raggedright\arraybackslash}p{0.17\textwidth}|>{\raggedright\arraybackslash}p{0.55\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|>{\raggedright\arraybackslash}p{0.10\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-37.1 (Create Help Request)} & The system shall allow a learner to create a help request. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-37.2 (Help Recipients)} & The system shall allow the learner to send the help request to friends, room participants, or concept learners. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-37.3 (Help Response)} & The system shall allow other learners to accept or decline a help request. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-37.4 (Cancel Help Request)} & The system shall allow the requester to cancel the help request. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-37.5 (Convert to Forum Post)} & The system shall allow a help request to be converted into a forum post for asynchronous discussion. & UI, Frontend, Backend & Not implemented \\ \hline
\textbf{FR-37.6 (Help Rate Limit)} & The system shall rate-limit help requests to prevent spam. & Backend & Not implemented \\ \hline
\end{longtable}
\endgroup
