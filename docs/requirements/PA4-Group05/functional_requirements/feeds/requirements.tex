\subsubsection*{US-13: View the List of Posts}
\par\medskip

\textbf{As a} university student, \textbf{I want to} view a list of posts within the selected forum, \textbf{so that} I can discover relevant discussions.
\par\medskip

\begingroup
\small
\renewcommand{\baselinestretch}{1.0}\selectfont
\renewcommand{\arraystretch}{1.15}

\begin{longtable}{|>{\raggedright\arraybackslash}p{0.14\textwidth}|>{\raggedright\arraybackslash}p{0.58\textwidth}|>{\raggedright\arraybackslash}p{0.09\textwidth}|>{\raggedright\arraybackslash}p{0.09\textwidth}|}
\hline
\textbf{FR} & \textbf{Requirement} & \textbf{Category} & \textbf{Status} \\ \hline
\endhead
\textbf{FR-13.1 (Post List Display)} & The system shall display a list of posts for the selected university, major, or course forum. & UI, Frontend, Backend & Complete \\ \hline
\textbf{FR-13.2 (Post Summary)} & The system shall show the post title, author, post type, timestamp, vote score, comment count, and solved status in the list. & UI, Frontend & Complete \\ \hline
\textbf{FR-13.4 (Post Sorting)} & The system shall support sorting posts by newest, top trending, most voted, and most discussed. & UI, Frontend, Backend & Complete \\ \hline
\textbf{FR-13.5 (Open Post)} & The system shall allow users to open a post from the list to view its full details. & UI, Frontend & Complete \\ \hline
\end{longtable}
